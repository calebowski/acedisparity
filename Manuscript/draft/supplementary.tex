\begin{figure}[H]
  \centering
  \includegraphics[width=\textwidth]{figures/continuous_boxplot_asymmetrical.pdf}
  \caption{\textbf{Continuous trait disparity error under asymmetrical trees}. Relative disparity error returned from pre-ordination point ASE (ancestral state estimation), pre-ordination distribution ASE, post-ordination point ASE, post-ordination distribution ASE and no ASE. These disparity errors were calculated across three evolutionary models (Brownian motion [BM], BM+trend and Ornstein-Uhlenbeck [OU]), five simulated fossil sampling densities (100\%, 50\%, 15\%, 5\% and 0\%) using the tip-only scheme and three disparity metrics (Sum of variances, sum of quantiles and mean pairwise distance). Values closest to 0 (dashed horizontal line) were most accurate; negative errors are under-estimates and positive are over-estimates.}
  \label{fig:Asymmetrical trees continuous boxplot}
\end{figure}

\begin{figure}[H]
  \centering
  \includegraphics[width=\textwidth]{figures/discrete_boxplot_assymetrical_50t_trees.pdf}
  \caption{\textbf{Discrete trait disparity error under asymmetrical trees}. Relative disparity error across ancestral state estimation methods and fossil sampling densities for discrete traits simulated on asymmetrical phylogenies. Results are shown for the tip-only fossil sampling scheme and are calculated across the transition rates and disparity metrics used in the main analysis. Values closer to zero indicate more accurate recovery of the true disparity. Negative values indicate underestimation and positive values indicate overestimation.} 
  \label{fig:Asymmetrical trees discrete boxplot}
\end{figure}
